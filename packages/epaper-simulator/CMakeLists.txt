cmake_minimum_required(VERSION 3.28)
project(epaper_simulator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(spring_gcc_options INTERFACE)
target_compile_options(spring_gcc_options INTERFACE
  -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow
  -Wnon-virtual-dtor -Wold-style-cast -Wcast-align -Wunused
  -Woverloaded-virtual -Wnull-dereference -Wdouble-promotion
  -Wformat=2 -Wimplicit-fallthrough -Werror)
add_subdirectory(../esp32p4/components/ui_core ui_core)
add_executable(epaper_simulator main.cpp)
target_link_libraries(epaper_simulator PRIVATE ui_core spring_gcc_options)
add_executable(router_test router_test.cpp)
target_link_libraries(router_test PRIVATE ui_core spring_gcc_options)
add_executable(no_hardware_test no_hardware_test.cpp)
target_link_libraries(no_hardware_test PRIVATE ui_core spring_gcc_options)
add_executable(snapshot_test snapshot_test.cpp)
target_link_libraries(snapshot_test PRIVATE ui_core spring_gcc_options)
add_executable(pattern_test pattern_test.cpp)
target_link_libraries(pattern_test PRIVATE ui_core spring_gcc_options)
add_executable(utf8_test utf8_test.cpp)
target_link_libraries(utf8_test PRIVATE ui_core spring_gcc_options)
enable_testing()
add_test(NAME render_clock COMMAND epaper_simulator ${CMAKE_CURRENT_BINARY_DIR}/clock.pbm)
add_test(NAME router_and_bounds COMMAND router_test)
add_test(NAME render_without_hardware COMMAND no_hardware_test ${CMAKE_CURRENT_BINARY_DIR}/route_)
add_test(NAME render_snapshots COMMAND snapshot_test)
add_test(NAME render_patterns COMMAND pattern_test ${CMAKE_CURRENT_BINARY_DIR}/pattern_)
add_test(NAME render_utf8 COMMAND utf8_test)
